# ============================================================
# PYTORCH INSTALLATION
# ============================================================

# For CPU-only systems / systems without NVIDIA GPU:
# Run after installing requirements.txt:
#
# python -m pip install torch torchvision torchaudio
#
# YOLO will run using CPU processing.


# ============================================================
# NVIDIA GPU USERS
# ============================================================

# First install the requirements normally:
#
# python -m pip install -r requirements.txt
#
# Then remove the CPU version of PyTorch:
#
# python -m pip uninstall torch torchvision torchaudio -y
#
# Install the CUDA 12.8 version of PyTorch:
#
# python -m pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu128
#
# Verify CUDA and GPU:
#
# python -c "import torch; print('PyTorch:', torch.__version__); print('CUDA available:', torch.cuda.is_available()); print('GPU:', torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'Not detected')"
#
# Expected output:
#
# PyTorch: 2.x.x+cu128
# CUDA available: True
# GPU: NVIDIA GeForce RTX 4050 Laptop GPU
#
# (The GPU name will be different depending on your NVIDIA GPU.)
